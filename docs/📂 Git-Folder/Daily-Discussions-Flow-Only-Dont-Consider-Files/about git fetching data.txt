function wrap(text, width, prefix,    result, line) {
  while (length(text) > width) {
    for (i = width; i > 0; i--) {
      if (substr(text, i, 1) == " ") {
        line = substr(text, 1, i);
        text = substr(text, i + 1);
        break;
      }
    }
    if (i == 0) {
      line = substr(text, 1, width);
      text = substr(text, width + 1);
    }
    result = result prefix line "\n";
  }
  return result prefix text;
}
