# --- Branch creation and setup --- ✅

# Make sure you're on working-code first 🧐
git checkout working-code

# Create and switch to new branch 🚀
git checkout -b feature/password-policy

# Push new branch to remote so it's tracked 🔄
git push -u origin feature/password-policy


# --- After development and testing --- ✅

# Switch back to working-code 🔙
git checkout working-code

# Merge the feature branch into working-code 🔀
git merge feature/password-policy

# Push merged changes to remote 🌍
git push origin working-code


# --- Cleanup (optional, after successful merge) --- 🧹

# Delete the local feature branch 🗑️
git branch -d feature/password-policy

# Delete the remote feature branch 🚮
git push origin --delete feature/password-policy


# --- Rollback / Undo if something goes wrong --- ⚠️

# 1. If merge is not pushed yet, reset working-code to previous commit ⏪
git reset --hard HEAD~1

# 2. If merge is already pushed to GitHub, revert the merge commit ⏪
#    (replace <merge_commit_sha> with the actual SHA from git log) 🔍
git revert -m 1 <merge_commit_sha>

# Push rollback changes to remote 🔁
git push origin working-code
