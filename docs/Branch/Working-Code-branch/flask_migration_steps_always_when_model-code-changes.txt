# =============================================
# Flask-Migrate Commands after Editing models.py
# =============================================

# 1. (Run only once per project, not needed again if migrations/ already exists)
flask db init
# → Creates the 'migrations/' directory and sets up migration environment.

# 2. Generate migration script after changes to models.py
flask db migrate -m "Describe your model changes here"
# → Detects model changes and creates a new migration script inside migrations/versions/

# 3. Apply the migration to the database
flask db upgrade
# → Applies the schema changes to the actual database (e.g., SQLite)

🔧 Database Migration for Enforcing Password History Policy

# 🏗️ Generate a new migration script to reflect changes in SQLAlchemy models
# This command scans your models.py and creates a new migration file under /migrations/versions
flask db migrate -m "Add password history"

# 📦 Apply the generated migration to your actual database
# This executes the upgrade operations written in the migration file (like creating/modifying tables)
flask db upgrade

# 🚀 Start the Flask development server (run the app locally on http://127.0.0.1:5000)
flask run
