# --- Development Mode (PowerShell syntax) ---
$env:FLASK_APP="app.py"                # Set the Flask application file
$env:FLASK_ENV="development"           # Enable debug mode
flask run                              # Run the app locally with debug features

# --- Production Mode (PowerShell syntax) ---
$env:FLASK_APP="app.py"                # Set the Flask application file
$env:FLASK_ENV="production"            # Disable debug mode for better performance
flask run                              # Run the app locally (not recommended for real production)
